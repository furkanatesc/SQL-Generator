**Goal:** Give the 29.3 Oracle adapter a real local-Docker test harness — an Oracle image, a seed schema, a Python seed applier, and a dedicated CI job — so the `oracle`-marked integration tests run live, and close TECH-DEBT §21.4 with a Docker-free bridge→low-level unit test.

## Global Constraints

- **No new execution behavior.** The 29.3 modules `oracle_adapter.py`, `oracle_connection_resolver.py`, `oracle_execution_adapter.py` are **not** modified (a docstring touch is the only allowed change). This is a harness sprint.
- **Driver isolation preserved.** Any new module must **not** import `oracledb` (or `cx_Oracle`, `sqlalchemy`) at module top level — `oracledb` imports lazily, inside functions only.
- **`oracle_seed.py` is NOT added to `app/evaluation/__init__.py`** — harness-only, keeps package surface clean (mirrors resolver/execution-adapter).
- **Oracle folds unquoted identifiers to UPPERCASE** — seed tables are `CUSTOMERS`/`ORDERS`/`ORDER_ITEMS`/`PRODUCT_VARIANTS`/`VARIANT_STOCK`; result column keys come back uppercase. Tests assert uppercase.
- **Seed contains no PL/SQL** — plain `CREATE TABLE` + `INSERT` only, `;`-terminated, so `split_statements` stays trivial.
- **Image pinned** to `gvenzl/oracle-free:23-slim-faststart`; env `ORACLE_PASSWORD=sqlgen`, `APP_USER=sqlgen`, `APP_USER_PASSWORD=sqlgen`; service `FREEPDB1`, port `1521` — all matching the 29.3 resolver defaults.
- **Commit messages:** plain, no Claude/co-author trailer. Prefix `feat(29.4):` / `test(29.4):` / `ci(29.4):` / `docs(29.4):`.
- **Before running the full suite:** ensure venv `fastapi` is pinned to `0.115.12` (see memory `sqlgen-fastapi-pin-drift`) or the `api_surface` test fails spuriously.

```python
"""Sprint 29.4 — Oracle integration-seed applier (harness only).

Pure statement splitter + DB-API seed applier used by the Oracle CI job and
docker-compose flow. NOT wired into any runtime path and NOT re-exported from
``app.evaluation``. The ``oracledb`` driver is imported lazily inside ``main``
only, preserving the 29.3 driver-isolation posture.
"""

import os
import sys
from typing import Optional, Sequence, Tuple

ORACLE_SEED_VERSION = "v1"

_DEFAULT_SEED_PATH = os.path.join("tests", "fixtures", "oracle", "seed.sql")


def split_statements(sql: str) -> Tuple[str, ...]:
    """Split a plain DDL+INSERT script into ``;``-terminated statements.

    Strips ``--`` line comments and blank lines. No PL/SQL support (the seed
    contains none), so a simple split on ``;`` is correct and deterministic.
    """
    lines = []
    for raw in sql.splitlines():
        stripped = raw.strip()
        if not stripped or stripped.startswith("--"):
            continue
        lines.append(raw)
    joined = "\n".join(lines)
    out = []
    for chunk in joined.split(";"):
        stmt = chunk.strip()
        if stmt:
            out.append(stmt)
    return tuple(out)


def apply_seed(connection, sql: str) -> int:
    """Execute each statement of ``sql`` on ``connection`` and commit once.

    ``connection`` is an already-open DB-API connection (injected). Returns the
    number of statements applied.
    """
    statements = split_statements(sql)
    cursor = connection.cursor()
    try:
        for stmt in statements:
            cursor.execute(stmt)
    finally:
        cursor.close()
    connection.commit()
    return len(statements)


def main(argv: Optional[Sequence[str]] = None) -> int:
    """CI entry-point: connect from ORACLE_TEST_* env and apply the seed file."""
    argv = list(sys.argv[1:] if argv is None else argv)
    seed_path = argv[0] if argv else _DEFAULT_SEED_PATH

    import oracledb  # noqa: WPS433 - runtime-only import, keeps driver isolation

    from app.evaluation.oracle_connection_resolver import (
        resolve_local_docker_connection,
    )

    conn_info = resolve_local_docker_connection()
    if conn_info is None:
        print("oracle_seed: no local Docker Oracle connection resolved", file=sys.stderr)
        return 1

    with open(seed_path, "r", encoding="utf-8") as fh:
        sql = fh.read()

    connection = oracledb.connect(
        user=conn_info.user,
        password=conn_info.password,
        dsn=f"{conn_info.host}:{conn_info.port}/{conn_info.service_name}",
    )
    try:
        applied = apply_seed(connection, sql)
    finally:
        connection.close()
    print(f"oracle_seed: applied {applied} statements from {seed_path}")
    return 0


if __name__ == "__main__":  # pragma: no cover
    raise SystemExit(main())
```

def _local_docker_oracle_available():
    try:
        import oracledb
    except ImportError:
        return False
    conn = resolve_local_docker_connection()
    if conn is None:
        return False
    try:
        db = oracledb.connect(
            user=conn.user, password=conn.password,
            dsn=f"{conn.host}:{conn.port}/{conn.service_name}",
            tcp_connect_timeout=2,
        )

def _request(sql, timeout=5.0, max_rows=1000):
    cfg = SQLDatabaseExecutionConfig(
        dialect=SQLDatabaseDialect.ORACLE, timeout_seconds=timeout,
        max_rows=max_rows, execution_mode=SQLExecutionMode.READ_ONLY,
    )
    return SQLDatabaseExecutionRequest(
        case_id="c1", sql=sql, dialect=SQLDatabaseDialect.ORACLE,
        fixture_ref=None, connection_ref="local_docker", config=cfg,
    )


def test_three_table_join_returns_named_rows(adapter):
    sql = (
        "SELECT c.name AS customer, o.status AS status, i.product_name AS product "
        "FROM customers c "
        "JOIN orders o ON o.customer_id = c.id "
        "JOIN order_items i ON i.order_id = o.id "

        "ORDER BY i.id"
