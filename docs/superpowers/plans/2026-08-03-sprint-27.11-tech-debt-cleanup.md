```python
import ast
import pathlib
import subprocess
import sys

_BACKEND_DIR = pathlib.Path(__file__).resolve().parents[2]
_ALLOWED_APP_PREFIXES = ("app.errors",)

import sys
before = set(sys.modules)
import app.errors  # noqa: F401
newly = set(sys.modules) - before
allowed_prefixes = {allowed_prefixes!r}

def test_errors_package_imports_only_stdlib():

        capture_output=True, text=True, cwd=str(_BACKEND_DIR))

    leaked_app, drivers = ast.literal_eval(result.stdout.strip())
    assert leaked_app == [], (
```

```python
import json
from pathlib import Path

_CONTRACTS_DIR = Path(__file__).resolve().parent / "contracts"
_HTTP_VERBS = {"get", "post", "put", "patch", "delete", "options", "head", "trace"}


def _snapshot_api_routes() -> set:
    """OpenAPI snapshot'indaki (METHOD_UPPER, path) ciftleri — tek kaynak."""
    snap = json.loads((_CONTRACTS_DIR / "openapi_snapshot.json").read_text(encoding="utf-8"))
    return {(method.upper(), path)
            for path, item in snap["paths"].items()
            for method in item if method in _HTTP_VERBS}
```

- [ ] **Step 3: FRAMEWORK_ROUTES'u ampirik belirle** — Geçici olarak testi şu şekilde koş: `discovered` seti ile `_snapshot_api_routes()` farkını yazdır. `backend/`'den:

```bash
./venv/Scripts/python.exe -c "from app.main import app; import json; from pathlib import Path; \
snap=json.loads(Path('tests/contracts/openapi_snapshot.json').read_text('utf-8')); \
verbs={'get','post','put','patch','delete','options','head','trace'}; \
sr={(m.upper(),p) for p,i in snap['paths'].items() for m in i if m in verbs}; \
dr={(list(r.methods)[0].upper() if getattr(r,'methods',None) else 'GET', r.path) for r in app.routes if hasattr(r,'path')}; \
print(sorted(dr - sr))"
```
